// one basis function of the DFT, swept
const N = 1024, sr = 44100;
export function render(ctx, t) {
  const k = Math.floor(t * 8) % 32;
  const re = new Float32Array(N);
  const im = new Float32Array(N);
  for (let n = 0; n < N; n++) {
    const w = 2 * Math.PI * k * n / N;
    re[n] = Math.cos(w);
    im[n] = -Math.sin(w);
  }
  plot(ctx, re, { label: `re k=${k}` });
  plot(ctx, im, { label: `im k=${k}` });
  return tone(k * sr / N, 0.4);
}

// magnitude spectrum of a windowed frame
import { hann, fft, mag } from './dsp.js';
export function render(ctx, buf, { N = 2048 }) {
  const frame = buf.slice(0, N).map(hann(N));
  const X = fft(frame);
  const dB = mag(X).map((m) => 20 * Math.log10(m));
  bars(ctx, dB, { peakColor: '#F2C230' });
}

// stereo phase portrait
export function frame(L, R, ctx) {
  ctx.begin({ fade: 0.08 });
  for (let i = 0; i < L.length; i++) {
    const x = (L[i] + R[i]) / SQRT2;
    const y = (L[i] - R[i]) / SQRT2;
    ctx.point(x, y, hue(i));
  }
  ctx.grid({ rings: 3 });
  return ctx.end();
}
